cmake_minimum_required(VERSION 3.22)
project(gdal4k C CXX)
option(GDAL4K_ENABLE_HTTP "Enable the isolated optional HTTP/CURL input profile" OFF)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gdal4k-archiver.txt" "${CMAKE_AR}\n")
if(VCPKG_INSTALLED_DIR AND VCPKG_TARGET_TRIPLET)
    set(GDAL4K_DEPENDENCIES "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}")
endif()
if(GDAL4K_DEPENDENCIES)
    list(PREPEND CMAKE_PREFIX_PATH "${GDAL4K_DEPENDENCIES}")
    # libtiff defaults to a static Apple framework on iOS. Both layouts must be
    # embedded as archives; consumers must never link a separate TIFF framework.
    if(EXISTS "${GDAL4K_DEPENDENCIES}/lib/libtiff.a")
        set(GDAL4K_TIFF "${GDAL4K_DEPENDENCIES}/lib/libtiff.a")
        set(GDAL4K_TIFF_INCLUDE "${GDAL4K_DEPENDENCIES}/include")
    elseif(EXISTS "${GDAL4K_DEPENDENCIES}/lib/tiff.framework/tiff")
        set(GDAL4K_TIFF "${GDAL4K_DEPENDENCIES}/lib/tiff.framework/tiff")
        set(GDAL4K_TIFF_INCLUDE "${GDAL4K_DEPENDENCIES}/lib/tiff.framework/Headers")
    else()
        message(FATAL_ERROR "The pinned static TIFF dependency is missing")
    endif()
    set(TIFF_INCLUDE_DIR "${GDAL4K_TIFF_INCLUDE}" CACHE PATH "" FORCE)
    set(TIFF_LIBRARY_RELEASE "${GDAL4K_TIFF}" CACHE FILEPATH "" FORCE)
    set(TIFF_LIBRARY_DEBUG "${GDAL4K_TIFF}" CACHE FILEPATH "" FORCE)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gdal4k-static-dependencies.txt" "${GDAL4K_TIFF}\n")
    # Never silently pick a producer machine's system/Homebrew GDAL.
    find_package(GDAL CONFIG REQUIRED PATHS "${GDAL4K_DEPENDENCIES}/share/gdal" NO_DEFAULT_PATH)
else()
    find_package(GDAL CONFIG REQUIRED)
endif()
if(GDAL4K_DEPENDENCIES)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(GDAL4K_RESOURCE_CPP "${CMAKE_CURRENT_BINARY_DIR}/gdal4k_resources.cpp")
    if(EMSCRIPTEN)
        set(GDAL4K_RESOURCE_OPTIONS --web)
    endif()
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../gdal4k-engine/tools/embed-data.py"
        "${GDAL4K_DEPENDENCIES}" "${GDAL4K_RESOURCE_CPP}" ${GDAL4K_RESOURCE_OPTIONS} COMMAND_ERROR_IS_FATAL ANY)
else()
    set(GDAL4K_RESOURCE_CPP "${CMAKE_CURRENT_BINARY_DIR}/gdal4k_resources.cpp")
    file(WRITE "${GDAL4K_RESOURCE_CPP}" "void gdal4k_install_resources() {}\nconst char* gdal4k_data_directory() { return \"\"; }\nconst char* gdal4k_license_notices() { return \"Developer runtime\"; }\n")
endif()
if(EMSCRIPTEN)
    add_executable(gdal4k gdal4k.cpp web-inputs.cpp web-outputs.cpp "${GDAL4K_RESOURCE_CPP}")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(GDAL4K_SERIAL_VIEWSHED "${CMAKE_CURRENT_BINARY_DIR}/viewshed-serial.cpp")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../gdal4k-engine/tools/serial-viewshed.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/serial-viewshed" "${GDAL4K_SERIAL_VIEWSHED}" COMMAND_ERROR_IS_FATAL ANY)
    target_sources(gdal4k PRIVATE "${GDAL4K_SERIAL_VIEWSHED}")
    target_include_directories(gdal4k PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/serial-viewshed")
    set_target_properties(gdal4k PROPERTIES SUFFIX ".mjs")
    target_compile_options(gdal4k PRIVATE -fexceptions)
    target_link_options(gdal4k PRIVATE
        -fexceptions --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1
        -sEXPORT_NAME=createGdal4kModule -sENVIRONMENT=web,worker
        -sALLOW_MEMORY_GROWTH=1 -sINITIAL_HEAP=1048576 -sMAXIMUM_MEMORY=2147483648
        -sSTACK_SIZE=1048576 -sFILESYSTEM=1 -sASYNCIFY=1 -sASYNCIFY_STACK_SIZE=65536
        "-sEXPORTED_FUNCTIONS=['_gdal4k_binary_version','_gdal4k_execute_binary','_gdal4k_binary_json','_gdal4k_binary_data','_gdal4k_binary_size','_gdal4k_binary_destroy','_gdal4k_create','_gdal4k_execute','_gdal4k_execute_controlled','_gdal4k_control_create','_gdal4k_control_cancel','_gdal4k_control_snapshot','_gdal4k_control_destroy','_gdal4k_free','_gdal4k_destroy','_malloc','_free']"
        "-sEXPORTED_RUNTIME_METHODS=['ccall','UTF8ToString','stringToNewUTF8','HEAPU8']")
else()
    add_library(gdal4k SHARED gdal4k.cpp "${GDAL4K_RESOURCE_CPP}")
    add_library(gdal4k_static STATIC gdal4k.cpp "${GDAL4K_RESOURCE_CPP}")
    target_compile_features(gdal4k_static PRIVATE cxx_std_17)
    target_include_directories(gdal4k_static PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(gdal4k_static PRIVATE GDAL4K_BUILD GDAL4K_STATIC)
    target_link_libraries(gdal4k_static PRIVATE GDAL::GDAL)
    set_target_properties(gdal4k_static PROPERTIES OUTPUT_NAME gdal4k-bridge POSITION_INDEPENDENT_CODE ON)
    if(WIN32)
        set_target_properties(gdal4k PROPERTIES PREFIX "")
        if(MINGW)
            target_link_options(gdal4k PRIVATE -static-libgcc -static-libstdc++ -Wl,-Bstatic -lwinpthread -Wl,-Bdynamic)
        endif()
    endif()
    if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
        if(DEFINED ENV{JAVA_HOME})
            file(TO_CMAKE_PATH "$ENV{JAVA_HOME}" GDAL4K_JAVA_HOME)
        else()
            find_package(JNI REQUIRED COMPONENTS JVM)
        endif()
        if(GDAL4K_JAVA_HOME)
            find_path(GDAL4K_JNI_INCLUDE jni.h PATHS "${GDAL4K_JAVA_HOME}/include" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
            find_path(GDAL4K_JNI_PLATFORM jni_md.h PATHS "${GDAL4K_JAVA_HOME}/include/linux" "${GDAL4K_JAVA_HOME}/include/win32" "${GDAL4K_JAVA_HOME}/include/darwin" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
            set(JNI_INCLUDE_DIRS "${GDAL4K_JNI_INCLUDE};${GDAL4K_JNI_PLATFORM}")
        endif()
        target_sources(gdal4k PRIVATE jni.cpp)
        target_include_directories(gdal4k PRIVATE ${JNI_INCLUDE_DIRS})
    elseif(ANDROID)
        target_sources(gdal4k PRIVATE jni.cpp)
        target_link_libraries(gdal4k PRIVATE log)
        target_link_options(gdal4k PRIVATE "-Wl,-z,max-page-size=16384")
    endif()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(gdal4k PRIVATE -static-libgcc -static-libstdc++ "-Wl,--exclude-libs,ALL")
elseif(ANDROID)
    target_link_options(gdal4k PRIVATE "-Wl,--exclude-libs,ALL")
endif()
target_compile_features(gdal4k PRIVATE cxx_std_17)
target_include_directories(gdal4k PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(gdal4k PRIVATE GDAL4K_BUILD)
target_link_libraries(gdal4k PRIVATE GDAL::GDAL)
if(GDAL4K_ENABLE_HTTP)
    if(EMSCRIPTEN OR NOT GDAL4K_DEPENDENCIES)
        message(FATAL_ERROR "Native HTTP requires the owned pinned optional CURL profile")
    endif()
    find_package(CURL CONFIG REQUIRED PATHS "${GDAL4K_DEPENDENCIES}/share/curl" NO_DEFAULT_PATH)
    foreach(target gdal4k gdal4k_static)
        target_compile_definitions(${target} PRIVATE GDAL4K_ENABLE_HTTP)
        target_link_libraries(${target} PRIVATE CURL::libcurl)
    endforeach()
    add_executable(gdal4k_http_probe http-cache-probe.cpp)
    target_compile_features(gdal4k_http_probe PRIVATE cxx_std_17)
    target_include_directories(gdal4k_http_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(gdal4k_http_probe PRIVATE GDAL4K_ENABLE_HTTP)
    target_link_libraries(gdal4k_http_probe PRIVATE GDAL::GDAL CURL::libcurl)
endif()
if(GDAL4K_DEPENDENCIES)
    # GDAL's FindJXL records libjxl.a alone. Resolve the codec's complete static closure.
    find_package(hwy CONFIG REQUIRED)
    find_package(unofficial-brotli CONFIG REQUIRED)
    find_package(lcms2 CONFIG REQUIRED)
    find_library(GDAL4K_JXL_CMS NAMES jxl_cms PATHS "${GDAL4K_DEPENDENCIES}/lib" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(gdal4k PRIVATE "${GDAL4K_JXL_CMS}" hwy::hwy
        unofficial::brotli::brotlienc unofficial::brotli::brotlidec unofficial::brotli::brotlicommon lcms2::lcms2)
    if(TARGET gdal4k_http_probe)
        target_link_libraries(gdal4k_http_probe PRIVATE "${GDAL4K_JXL_CMS}" hwy::hwy
            unofficial::brotli::brotlienc unofficial::brotli::brotlidec unofficial::brotli::brotlicommon lcms2::lcms2)
    endif()
endif()
set_target_properties(gdal4k PROPERTIES CXX_VISIBILITY_PRESET hidden)
install(TARGETS gdal4k LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES gdal4k.h DESTINATION include)
